    // A variável 'products' é carregada globalmente a partir de dados.js

        function createProductCardHTML(product) {
            const productId = product.name.replace(/\s+/g, '-');
            const specsHtml = product.specs.map(spec => `
                <li class="flex items-start">
                    <strong class="w-1/3">${spec.label}:</strong>
                    <span class="ml-1.5 flex-1">${spec.value}</span>
                </li>
            `).join('');

            const imageTag = `<img src="${product.imageSrc}" alt="Imagem de ${product.name}" class="rounded-lg mx-auto h-auto cursor-pointer" style="max-width: 300px;">`;

            // Lógica de classes refeita para maior clareza e robustez.
            // Garante que 'colorlight-card' seja adicionado a QUALQUER produto colorlight, incluindo receivers.
            const cardClasses = ['card', 'cursor-pointer'];
            if (product.type === 'receiver') {
                cardClasses.push('receiver-card');
            }
            if (product.category.includes('cat-colorlight')) {
                cardClasses.push('colorlight-card');
            }

            return `
                <div class="product-item w-full md:w-1/2 xl:w-1/3 px-4 mb-8" data-category="${product.category}">
                    <div class="${cardClasses.join(' ')}" data-product-name="${product.name}" data-product-type="${product.type}" data-image-src="${product.imageSrc}" data-technical-file-url="${product.technicalFileUrl || ''}">
                         <div class="absolute top-2 right-2 z-10">
                            <label for="compare-${productId}" class="compare-checkbox-container flex items-center space-x-2 cursor-pointer p-2 bg-gray-900 bg-opacity-60 rounded-full hover:bg-opacity-80 transition-all" title="Selecionar para comparar">
                                <input type="checkbox" id="compare-${productId}" data-product-name="${product.name}" class="compare-checkbox h-4 w-4 rounded">
                            </label>
                        </div>
                        <div class="p-6 card-content">
                            <h2 class="text-2xl font-bold text-gray-100 text-center mb-4">${product.name}</h2>
                            <div class="mb-4 flex justify-center items-center" style="height: ${product.type === 'receiver' ? '300px' : '150px'};">
                                ${imageTag}
                            </div>
                            <h3 class="font-semibold text-lg mb-3 text-gray-200 border-b border-gray-600 pb-2">Especificações Principais</h3>
                            <ul class="space-y-3 text-gray-300 text-sm">${specsHtml}</ul>
                            ${product.recommendation ? `
                            <div class="mt-6">
                                <h4 class="font-semibold text-md mb-2 text-gray-200">Recomendação</h4>
                                <p class="text-sm text-gray-400">${product.recommendation}</p>
                            </div>
                            ` : ''}
                        </div>
                    </div>
                </div>
            `;
        }
        
        function initializeGrids() {
            if (typeof products === 'undefined') {
                console.error("O banco de dados de produtos (products) não foi encontrado. Verifique se o arquivo dados.js está sendo carregado corretamente.");
                return;
            }
            products.forEach(product => {
                const cardHTML = createProductCardHTML(product);
                if (product.type === 'processadora' && productGridProcessadoras) {
                    productGridProcessadoras.insertAdjacentHTML('beforeend', cardHTML);
                } else if (product.type === 'sender' && productGridSenders) {
                    productGridSenders.insertAdjacentHTML('beforeend', cardHTML);
                } else if (product.type === 'receiver' && productGridReceivers) {
                    productGridReceivers.insertAdjacentHTML('beforeend', cardHTML);
                } else if (product.type === 'tbs' && productGridTBs) {
                    productGridTBs.insertAdjacentHTML('beforeend', cardHTML);
                }
            });
             // Adiciona o listener para o efeito de brilho após os cards serem criados
            document.querySelectorAll('.card').forEach(card => {
                card.onmousemove = e => {
                    const rect = card.getBoundingClientRect();
                    const x = e.clientX - rect.left;
                    const y = e.clientY - rect.top;
                    card.style.setProperty('--mouse-x', `${x}px`);
                    card.style.setProperty('--mouse-y', `${y}px`);
                };
            });
        }

        function setupFilter(filterContainer, gridContainer, titleElement, defaultTitle) {
             if (!filterContainer) return;
             filterContainer.addEventListener('click', (event) => {
                const targetButton = event.target.closest('.filter-btn');
                if (!targetButton) return;

                filterContainer.querySelectorAll('.filter-btn').forEach(btn => btn.classList.remove('active'));
                targetButton.classList.add('active');

                const filter = targetButton.getAttribute('data-filter');
                if (titleElement) {
                    titleElement.textContent = targetButton.textContent === 'Todos' ? defaultTitle : targetButton.textContent;
                }
                
                gridContainer.querySelectorAll('.product-item').forEach(item => {
                    const itemCategories = item.getAttribute('data-category').split(' ');
                    const shouldBeVisible = (filter === 'all' || itemCategories.includes(filter));
                    item.classList.toggle('hidden', !shouldBeVisible);
                });
            });
        }
        
        // FUNÇÃO ATUALIZADA: para configurar o filtro de dropdown
        function setupDropdownFilter(dropdownElement, gridContainer, titleElement, defaultTitle) {
            if (!dropdownElement) return;
        
            // Garante que o gradiente esteja presente na inicialização, já que "Todos" é o padrão
            dropdownElement.classList.add('active-gradient');

            dropdownElement.addEventListener('change', (event) => {
                const filter = event.target.value; 
        
                if (titleElement) {
                    const selectedOptionText = event.target.options[event.target.selectedIndex].text;
                    titleElement.textContent = filter === 'all' ? defaultTitle : selectedOptionText;
                }
        
                gridContainer.querySelectorAll('.product-item').forEach(item => {
                    const itemCategories = item.getAttribute('data-category').split(' ');
                    const shouldBeVisible = (filter === 'all' || itemCategories.includes(filter));
                    item.classList.toggle('hidden', !shouldBeVisible);
                });

                // A classe 'active-gradient' permanece no dropdown, pois ele sempre está em um estado de filtro
                dropdownElement.classList.add('active-gradient'); 
            });
        }

        function updateCompareUI() {
            const count = selectedForComparison.length;
            compareBtn.textContent = `Comparar Itens (${count})`;
            compareBar.classList.toggle('visible', count >= 2);
        }
        
        function renderComparisonTable() {
            const productsToCompare = products.filter(p => selectedForComparison.includes(p.name));
            if (productsToCompare.length < 2) return;

            const allSpecLabels = ['Imagem', ...new Set(productsToCompare.flatMap(p => p.specs.map(s => s.label))), 'Ambiente Recomendado'];

            let tableHTML = '<table class="comparison-table">';
            tableHTML += '<thead><tr><th>Característica</th>';
            productsToCompare.forEach(p => { tableHTML += `<th>${p.name}</th>`; });
            tableHTML += '</tr></thead>';
            tableHTML += '<tbody>';
            allSpecLabels.forEach(label => {
                tableHTML += `<tr><td>${label}</td>`;
                productsToCompare.forEach(product => {
                    if (label === 'Imagem') {
                        // A imagem na tabela de comparação não terá borda gradiente por padrão
                        tableHTML += `<td><img src="${product.imageSrc}" alt="${product.name}" class="rounded-lg mx-auto max-w-[200px]"></td>`;
                    } else if (label === 'Ambiente Recomendado') {
                        if (product.recommendation) {
                            tableHTML += `<td>${product.recommendation}</td>`;
                        } else {
                            tableHTML += `<td>—</td>`;
                        }
                    } else {
                        const spec = product.specs.find(s => s.label === label);
                        tableHTML += `<td>${spec ? spec.value : '—'}</td>`;
                    }
                });
                tableHTML += '</tr>';
            });
            tableHTML += '</tbody></table>';

            comparisonTableContainer.innerHTML = tableHTML;
            document.body.style.overflow = 'hidden';
            comparisonView.style.display = 'block';
        }

        // FUNÇÃO ATUALIZADA: para exibir detalhes do produto (e seu equivalente) em um modal
        function showProductDetails(productName) {
            const product = products.find(p => p.name === productName);
            if (!product) return;

            const equivalentProduct = products.find(p => p.name === product.equivalentProduct);

            // Se não houver equivalente, mostra apenas o produto clicado
            if (!equivalentProduct) {
                const titleHtml = `<h2 class="text-3xl font-bold gradient-text text-center mb-6">${product.name}</h2>`;
                const productSpecsHtml = product.specs.map(spec => `
                    <li class="flex items-start">
                        <strong class="w-1/3">${spec.label}:</strong>
                        <span class="ml-1.5 flex-1">${spec.value}</span>
                    </li>
                `).join('');
                const contentHtml = `
                    <div class="flex flex-col md:flex-row items-start gap-6">
                        <div class="md:w-1/2 flex justify-center items-center">
                            <img src="${product.imageSrc}" alt="Imagem de ${product.name}" class="rounded-lg w-full h-auto">
                        </div>
                        <div class="md:w-1/2 text-gray-300">
                            <h3 class="font-semibold text-lg mb-3 text-gray-200 border-b border-gray-600 pb-2">Especificações</h3>
                            <ul class="space-y-2 text-sm">${productSpecsHtml}</ul>
                        </div>
                    </div>
                `;
                modalProductContent.innerHTML = titleHtml + contentHtml;
            } else {
                // Caso haja um equivalente, mostra ambos lado a lado
                const titleHtml = `<h2 class="text-3xl font-bold gradient-text text-center mb-6">${product.name} equivalente a ${equivalentProduct.name}</h2>`;

                const productSpecsHtml = product.specs.map(spec => `
                    <li class="flex items-start">
                        <strong class="w-1/3">${spec.label}:</strong>
                        <span class="ml-1.5 flex-1">${spec.value}</span>
                    </li>
                `).join('');
                
                const equivalentSpecsHtml = equivalentProduct.specs.map(spec => `
                    <li class="flex items-start">
                        <strong class="w-1/3">${spec.label}:</strong>
                        <span class="ml-1.5 flex-1">${spec.value}</span>
                    </li>
                `).join('');

                const contentHtml = `
                    <div class="flex flex-col md:flex-row items-start gap-6">
                        <!-- Produto Clicado -->
                        <div class="md:w-1/2 flex flex-col items-center">
                            <div class="image-wrapper mb-4 w-full flex justify-center items-center">
                                <!-- Reduzindo o tamanho da imagem -->
                                <img src="${product.imageSrc}" alt="Imagem de ${product.name}" class="rounded-lg h-auto max-w-[200px]">
                            </div>
                            <h3 class="font-semibold text-lg mb-3 text-gray-200 border-b border-gray-600 pb-2">Especificações</h3>
                            <ul class="space-y-2 text-sm text-gray-300 w-full">${productSpecsHtml}</ul>
                        </div>
                        
                        <!-- Divisor entre as colunas -->
                        <div class="md:w-px md:h-auto w-full h-px bg-gray-600 md:self-stretch my-4 md:my-0"></div>

                        <!-- Produto Equivalente -->
                        <div class="md:w-1/2 flex flex-col items-center">
                            <div class="image-wrapper mb-4 w-full flex justify-center items-center">
                                <!-- Reduzindo o tamanho da imagem -->
                                <img src="${equivalentProduct.imageSrc}" alt="Imagem de ${equivalentProduct.name}" class="rounded-lg h-auto max-w-[200px]">
                            </div>
                            <h3 class="font-semibold text-lg mb-3 text-gray-200 border-b border-gray-600 pb-2">Especificações</h3>
                            <ul class="space-y-2 text-sm text-gray-300 w-full">${equivalentSpecsHtml}</ul>
                        </div>
                    </div>
                `;

                modalProductContent.innerHTML = titleHtml + contentHtml;
            }
            
            productDetailModal.classList.remove('hidden');
            document.body.style.overflow = 'hidden'; // Evita scroll no body
        }

        // NOVO: Função para abrir o modal da imagem
        function showImageModal(imageSrc, technicalFileUrl) {
            modalFullImage.src = imageSrc;
            imageModal.classList.remove('hidden');
            document.body.style.overflow = 'hidden';

            // Adiciona a funcionalidade de clicar no botão "Ver Arquivo Técnico"
            if (technicalFileUrl) {
                technicalFileBtn.style.display = 'block';
                technicalFileBtn.onclick = () => window.open(technicalFileUrl, '_blank');
            } else {
                technicalFileBtn.style.display = 'none';
            }
        }

        // NOVO: Função para fechar o modal da imagem
        function closeImageModal() {
            imageModal.classList.add('hidden');
            document.body.style.overflow = '';
        }

        navMenu.addEventListener('click', (e) => {
            const parentLink = e.target.closest('.nav-link-parent');
            const pageLink = e.target.closest('.nav-link');

            if (parentLink) {
                e.preventDefault();
                const submenuId = parentLink.getAttribute('data-toggle');
                const submenu = document.getElementById(submenuId);
                const icon = parentLink.querySelector('.chevron-icon');
                submenu.classList.toggle('open');
                parentLink.classList.toggle('active');
                icon.classList.toggle('rotated');
                return;
            }

            if (pageLink) {
                e.preventDefault();
                document.querySelectorAll('#nav-menu .nav-link').forEach(l => l.classList.remove('active'));
                pages.forEach(p => p.classList.remove('active'));
                const pageId = pageLink.getAttribute('data-page');
                document.getElementById(pageId).classList.add('active');
                pageLink.classList.add('active');
                // Fechando o menu-container ao navegar
                body.classList.remove('menu-open');
            }
        });
        
        mainContent.addEventListener('click', (event) => {
            // VERIFICAÇÃO ADICIONADA: Se o clique foi no contêiner do checkbox,
            // ignoramos o restante da lógica para não abrir o modal.
            if (event.target.closest('.compare-checkbox-container')) {
                return; 
            }

            // Verifica se o clique foi em um card de produto.
            const clickedCard = event.target.closest('.card');
            if (clickedCard) {
                const productName = clickedCard.dataset.productName;
                const productType = clickedCard.dataset.productType;

                // Para os "Receivers" abre o modal de detalhes do produto.
                if (productType === 'receiver') {
                    showProductDetails(productName);
                    return;
                }
                
                // Para todos os outros tipos, abre o modal de imagem.
                const imageSrc = clickedCard.dataset.imageSrc;
                const technicalFileUrl = clickedCard.dataset.technicalFileUrl;
                showImageModal(imageSrc, technicalFileUrl);
            }
        });


        closeImageModalBtn.addEventListener('click', closeImageModal);


        mainContent.addEventListener('change', (event) => {
            if (event.target.classList.contains('compare-checkbox')) {
                const checkbox = event.target;
                const productName = checkbox.dataset.productName;
                const clickedCard = checkbox.closest('.card');
                if (!clickedCard) return;

                const clickedProduct = products.find(p => p.name === productName);
                if (!clickedProduct) return;
                
                const clickedProductType = clickedProduct.type;

                if (checkbox.checked) {
                    // Verifica se já existem itens selecionados e se são de uma categoria diferente.
                    if (selectedForComparison.length > 0) {
                        const firstSelectedName = selectedForComparison[0];
                        const firstSelectedProduct = products.find(p => p.name === firstSelectedName);
                        
                        if (firstSelectedProduct && firstSelectedProduct.type !== clickedProductType) {
                            // Se as categorias forem diferentes, limpa a seleção antiga.
                            // Desmarca todos os outros checkboxes.
                            document.querySelectorAll('.compare-checkbox:checked').forEach(cb => {
                                if (cb !== checkbox) {
                                    cb.checked = false;
                                }
                            });
                            // Limpa o array de comparação.
                            selectedForComparison = [];
                        }
                    }
                    
                    // Adiciona o novo item à lista (que pode ter sido limpa).
                    if (!selectedForComparison.includes(productName)) {
                        selectedForComparison.push(productName);
                    }
                } else {
                    // Remove o item se for desmarcado.
                    selectedForComparison = selectedForComparison.filter(name => name !== productName);
                }
                
                updateCompareUI();
            }
        });

        // Listener para fechar o modal de detalhes do produto
        closeDetailModalBtn.addEventListener('click', closeProductDetailsModal);


        compareBtn.addEventListener('click', renderComparisonTable);
        
        closeComparisonBtn.addEventListener('click', () => {
            comparisonView.style.display = 'none';
            document.body.style.overflow = '';
        });
        
        clearComparisonBtn.addEventListener('click', () => {
            selectedForComparison = [];
            document.querySelectorAll('.compare-checkbox:checked').forEach(checkbox => {
                checkbox.checked = false;
            });
            updateCompareUI();
        });

        // --- INICIALIZAÇÃO ---
        initializeGrids();
        
        // Chamada de inicialização para o filtro de Senders (sem alteração)
        setupFilter(filterButtonsSenders, productGridSenders, sendersTitle, 'Senders');
        
        // Chamada de inicialização para o NOVO filtro de Processadoras
        // Seleciona o elemento dropdown
        const filterDropdownProcessadoras = document.getElementById('filter-dropdown');
        // Chama a nova função de filtro, passando o dropdown como argumento
        setupDropdownFilter(filterDropdownProcessadoras, productGridProcessadoras, processadorasTitle, 'Processadoras');
        
        // Abre o submenu de Produtos por padrão
        const initialSubmenu = document.getElementById('submenu-produtos');
        const initialParent = document.querySelector('[data-toggle="submenu-produtos"]');
        if (initialSubmenu && initialParent) {
            // Não abre mais o submenu por padrão para dar foco ao dashboard
            // initialSubmenu.classList.add('open');
            // initialParent.classList.add('active');
            // initialIcon.classList.add('rotated');
        }
    });
